Data Loading & Preprocessing

Show Code 
# Paths
path <- "data"
# CPI (headline + food)
fh   <- to_monthly(fred_read(file.path(path, "CUSR0000SAF11.csv"))) # Food at Home
fa   <- to_monthly(fred_read(file.path(path, "CUSR0000SEFV.csv")))  # Food Away from Home
cpi  <- to_monthly(fred_read(file.path(path, "CPIAUCSL.csv")))      # Overall CPI
# Rates
dgs  <- to_monthly(fred_read(file.path(path, "DGS10.csv")))         # 10Y Treasury
ff   <- to_monthly(fred_read(file.path(path, "FEDFUNDS.csv")))      # Fed Funds

# ts objects (monthly)
fh_ts  <- ts(fh$value,  start = ym_start(fh),  frequency = 12)
fa_ts  <- ts(fa$value,  start = ym_start(fa),  frequency = 12)
cpi_ts <- ts(cpi$value, start = ym_start(cpi), frequency = 12)
dgs_ts <- ts(dgs$value, start = ym_start(dgs), frequency = 12)
ff_ts  <- ts(ff$value,  start = ym_start(ff),  frequency = 12)

# log-transform for CPI-like series
fh_log  <- log(fh_ts)
fa_log  <- log(fa_ts)
cpi_log <- log(cpi_ts)

Note. CPI series are modeled on the log scale (common in inflation work). Interest rates remain on levels.

as_monthly_ts <- function(x) {

  if (inherits(x, "ts")) {
    x <- stats::na.omit(x)
    ts(as.numeric(x), start = start(x), frequency = ifelse(frequency(x) > 1, frequency(x), 12))
  } else {
    x <- stats::na.omit(as.numeric(x))
    ts(x, frequency = 12)
  }
}

safe_arima_fit <- function(x, p, d, q, include_drift = (d == 1)) {
  fit <- tryCatch(
    Arima(x, order = c(p, d, q), include.drift = include_drift, method = "ML"),
    error = function(e) NULL
  )
  if (is.null(fit)) {
    fit <- tryCatch(
      Arima(x, order = c(p, d, q), include.drift = include_drift, method = "CSS-ML"),
      error = function(e) NULL
    )
  }
  fit
}

manual_arima_grid <- function(x, d = 1, pmax = 4, qmax = 4) {
  rows <- vector("list", (pmax + 1) * (qmax + 1))
  errs <- vector("list", length(rows))
  k <- 1
  for (p in 0:pmax) {
    for (q in 0:qmax) {
      fit <- safe_arima_fit(x, p, d, q)
      if (is.null(fit)) {
        rows[[k]] <- data.frame(p = p, d = d, q = q,
                                AIC = NA_real_, BIC = NA_real_, AICc = NA_real_,
                                status = "failed")
      } else {
        rows[[k]] <- data.frame(p = p, d = d, q = q,
                                AIC = AIC(fit),
                                BIC = BIC(fit),
                                AICc = if (!is.null(fit$aicc)) fit$aicc else NA_real_,
                                status = "ok")
      }
      k <- k + 1
    }
  }
  out <- do.call(rbind, rows)
  out <- out[order(out$BIC), ]
  rownames(out) <- NULL
  out
}
