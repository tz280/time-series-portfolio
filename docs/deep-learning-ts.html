Deep Learning Time Series

Introduction

This section applies deep learning methods (RNN, GRU, LSTM) to forecast inflation measures and compares them with traditional models. (e.g : Arima from Homework 3)

Key Design Choices:

  • Window Size: 12 months (captures yearly patterns and seasonality)
  • Forecast Horizon: 1-step ahead (next month prediction)
  • Train/Validation/Test Split: 70% / 15% / 15% (chronological)
  • Architecture: Single-layer networks with 64 units
  • Regularization: Dropout (0.2) + Early Stopping (patience=15)

fred_read <- function(path){
  read_csv(path, show_col_types = FALSE) |>
    rename(date = 1, value = 2) |>
    mutate(date = as.Date(date), value = suppressWarnings(as.numeric(value))) |>
    filter(!is.na(value))
}

to_monthly <- function(df, fun = mean){
  df |>
    mutate(ym = as.yearmon(date)) |>
    group_by(ym) |>
    summarize(value = fun(value, na.rm = TRUE), .groups = "drop") |>
    arrange(ym)
}

ym_start <- function(df) {
  ym <- df$ym[1]
  c(as.integer(format(ym, "%Y")), as.integer(format(ym, "%m")))
}

# Function to create windowed dataset for deep learning
create_dataset <- function(data, window_size = 12) {
  n <- length(data)
  X <- list()
  y <- numeric()
  
  for (i in 1:(n - window_size)) {
    X[[i]] <- data[i:(i + window_size - 1)]
    y[i] <- data[i + window_size]
  }
  
  # Convert to array with explicit float32 type
  X_array <- array(as.numeric(unlist(X)), dim = c(length(X), window_size, 1))
  y_array <- as.numeric(y)
  
  # Ensure float32
  storage.mode(X_array) <- "double"
  storage.mode(y_array) <- "double"
  
  list(X = X_array, y = y_array)
}

# Function to split data chronologically
split_data <- function(X, y, train_ratio = 0.7, val_ratio = 0.15) {
  n <- nrow(X)
  train_end <- floor(n * train_ratio)
  val_end <- floor(n * (train_ratio + val_ratio))
  
  list(
    X_train = X[1:train_end, , , drop = FALSE],
    y_train = y[1:train_end],
    X_val = X[(train_end + 1):val_end, , , drop = FALSE],
    y_val = y[(train_end + 1):val_end],
    X_test = X[(val_end + 1):n, , , drop = FALSE],
    y_test = y[(val_end + 1):n]
  )
}

# Function to build models
build_rnn <- function(window_size, units = 64) {
  model <- keras_model_sequential()
  model$add(keras$layers$Input(shape = as.integer(c(window_size, 1))))
  model$add(keras$layers$SimpleRNN(units = as.integer(units)))
  model$add(keras$layers$Dropout(rate = 0.2))
  model$add(keras$layers$Dense(units = 1L))
  return(model)
}

build_gru <- function(window_size, units = 64) {
  model <- keras_model_sequential()
  model$add(keras$layers$Input(shape = as.integer(c(window_size, 1))))
  model$add(keras$layers$GRU(units = as.integer(units)))
  model$add(keras$layers$Dropout(rate = 0.2))
  model$add(keras$layers$Dense(units = 1L))
  return(model)
}

build_lstm <- function(window_size, units = 64) {
  model <- keras_model_sequential()
  model$add(keras$layers$Input(shape = as.integer(c(window_size, 1))))
  model$add(keras$layers$LSTM(units = as.integer(units)))
  model$add(keras$layers$Dropout(rate = 0.2))
  model$add(keras$layers$Dense(units = 1L))
  return(model)
}

# Function to train model
train_model <- function(model, X_train, y_train, X_val, y_val, 
                       epochs = 100, batch_size = 32) {
  # Convert to TensorFlow tensors
  X_train_tf <- tensorflow::tf$constant(X_train, dtype = tensorflow::tf$float32)
  y_train_tf <- tensorflow::tf$constant(y_train, dtype = tensorflow::tf$float32)
  X_val_tf <- tensorflow::tf$constant(X_val, dtype = tensorflow::tf$float32)
  y_val_tf <- tensorflow::tf$constant(y_val, dtype = tensorflow::tf$float32)
  
  model$compile(
    optimizer = keras$optimizers$Adam(learning_rate = 0.001),
    loss = 'mse',
    metrics = list('mae')
  )
  
  early_stop <- keras$callbacks$EarlyStopping(
    monitor = 'val_loss',
    patience = as.integer(15),
    restore_best_weights = TRUE
  )
  
  history <- model$fit(
    X_train_tf, y_train_tf,
    epochs = as.integer(epochs),
    batch_size = as.integer(batch_size),
    validation_data = list(X_val_tf, y_val_tf),
    callbacks = list(early_stop),
    verbose = 0L
  )
  
  list(model = model, history = history)
}

# Function to evaluate and plot
evaluate_model <- function(model, X_test, y_test, data_full, window_size, 
                          series_name, model_name) {
  # Use model$predict directly (avoid forecast package conflict)
  X_test_tf <- tensorflow::tf$constant(X_test, dtype = tensorflow::tf$float32)
  pred_test <- model$predict(X_test_tf, verbose = 0L)
  pred_test <- as.vector(pred_test)
  
  # Calculate RMSE
  rmse <- sqrt(mean((y_test - pred_test)^2))
  mae <- mean(abs(y_test - pred_test))
  
  # Create data frame for plotting
  test_indices <- (length(data_full) - length(y_test) + 1):length(data_full)
  
  plot_df <- data.frame(
    index = test_indices,
    actual = y_test,
    predicted = pred_test
  ) %>%
    pivot_longer(cols = c(actual, predicted), 
                 names_to = "type", 
                 values_to = "value")
  
  # Plot
  p <- ggplot(plot_df, aes(x = index, y = value, color = type)) +
    geom_line(linewidth = 1) +
    labs(
      title = paste(model_name, "Forecast:", series_name),
      subtitle = sprintf("Test RMSE: %.6f | MAE: %.6f", rmse, mae),
      x = "Time Index",
      y = "Log Value",
      color = NULL
    ) +
    scale_color_manual(values = c("actual" = "black", "predicted" = "#2E86AB")) +
    theme_minimal() +
    theme(legend.position = "bottom")
  
  list(rmse = rmse, mae = mae, plot = p, predictions = pred_test)
}

# Multi-step forecasting function
evaluate_multi_step <- function(model, data_full, window_size, 
                               horizons = c(1, 3, 6, 12),
                               test_size = 24) {
  n <- length(data_full)
  test_start <- n - test_size + 1
  
  results <- list()
  
  for (h in horizons) {
    predictions <- numeric(test_size - h + 1)
    actuals <- numeric(test_size - h + 1)
    
    for (i in 1:(test_size - h + 1)) {
      start_idx <- test_start + i - 1
      current_window <- data_full[start_idx:(start_idx + window_size - 1)]
      
      if(any(is.na(current_window))) {
        predictions[i] <- NA
        actuals[i] <- data_full[start_idx + window_size + h - 1]
        next
      }
      
      for (step in 1:h) {
        X_input <- array(current_window, dim = c(1, window_size, 1))
        X_input_tf <- tensorflow::tf$constant(X_input, dtype = tensorflow::tf$float32)
        
        pred <- model$predict(X_input_tf, verbose = 0L)
        pred_value <- as.numeric(pred[1, 1])
        
        if(is.na(pred_value) || is.nan(pred_value) || is.infinite(pred_value)) {
          pred_value <- current_window[window_size]  
        }
        
        current_window <- c(current_window[-1], pred_value)
      }
      
      predictions[i] <- pred_value
      actuals[i] <- data_full[start_idx + window_size + h - 1]
    }
    
    valid_idx <- !is.na(predictions) & !is.na(actuals)
    
    if(sum(valid_idx) > 0) {
      rmse <- sqrt(mean((actuals[valid_idx] - predictions[valid_idx])^2))
      mae <- mean(abs(actuals[valid_idx] - predictions[valid_idx]))
    } else {
      rmse <- NA
      mae <- NA
    }
    
    results[[as.character(h)]] <- list(
      horizon = h,
      rmse = rmse,
      mae = mae,
      predictions = predictions,
      actuals = actuals
    )
  }
  
  return(results)
}

# Plot multi-step horizon performance
plot_horizon_performance <- function(horizon_results, series_name, model_name) {
  # Extract RMSE for each horizon
  horizon_df <- data.frame(
    horizon = sapply(horizon_results, function(x) x$horizon),
    rmse = sapply(horizon_results, function(x) x$rmse),
    mae = sapply(horizon_results, function(x) x$mae)
  )
  
  # Remove any NA or Inf values
  horizon_df <- horizon_df[is.finite(horizon_df$rmse), ]
  
  if(nrow(horizon_df) == 0) {
    return(ggplot() + 
             annotate("text", x = 0.5, y = 0.5, 
                     label = "No valid data for plotting", 
                     size = 6) +
             theme_void())
  }
  
  ggplot(horizon_df, aes(x = horizon, y = rmse)) +
    geom_line(linewidth = 1, color = "#2E86AB") +
    geom_point(size = 3, color = "#2E86AB") +
    geom_text(aes(label = sprintf("%.5f", rmse)), 
              vjust = -0.8, size = 3.5) +
    labs(
      title = paste(model_name, "- Forecast Horizon Performance"),
      subtitle = series_name,
      x = "Forecast Horizon (months ahead)",
      y = "RMSE"
    ) +
    scale_x_continuous(breaks = horizon_df$horizon) +
    expand_limits(y = 0) +
    theme_minimal() +
    theme(
      plot.title = element_text(face = "bold", size = 14),
      plot.subtitle = element_text(size = 11)
    )
}

Comparing traditional and deep learning approaches on CPI data revealed a nuanced story about model-data fit over raw complexity. Quantitatively, deep learning models achieved 12-51% lower test RMSE than ARIMA across all three series, with LSTM averaging 0.124 compared to ARIMA’s 0.159. However, this performance advantage came with important trade-offs and series-specific insights.

The most valuable lesson emerged from analyzing which models excel where. LSTM dominated on volatile, trending series (Food at Home: 0.091 RMSE), capturing non-linear inflation dynamics that ARIMA’s linear structure missed. Yet LSTM performed worst on the smoother Food Away series (0.160), where even ARIMA (0.137) outperformed it. This demonstrates that architectural sophistication can backfire when data patterns are simpler than the model assumes—GRU’s success on Food Away (0.133) confirms this. Meanwhile, RNN maintained remarkably consistent performance (0.111-0.141) across all series, revealing the value of balanced complexity.

Qualitatively, these differences matter for practical forecasting. LSTM’s superior accuracy on Food at Home enables more confident inventory planning, while its over-smoothed predictions on simpler series waste computational resources without improving decisions. ARIMA, though less accurate, offers immediate interpretability—coefficients directly translate to economic stories about momentum and shocks—making it invaluable when stakeholders need explainable forecasts.

The fundamental trade-off crystallized: ARIMA trains in seconds with clear parameters but underfits complex patterns; deep learning captures subtle non-linearities but requires minutes of training, careful tuning, and acts as a black box. For production forecasting, this suggests a hybrid strategy: deploy LSTM where accuracy justifies complexity (volatile CPI components), retain ARIMA where interpretability and speed matter (reporting, real-time updates), and use RNN as a robust baseline for model monitoring.

Ultimately, comparing these approaches taught me that forecasting is not about finding “the best model” but matching model capabilities to data characteristics and business constraints. The 20-50% RMSE improvements from deep learning are meaningful but must be weighed against interpretability loss and operational complexity—a trade-off that varies by use case, not dataset.

# Create multivariate dataset for deep learning
create_multivariate_dataset <- function(data_matrix, window_size = 12, target_col = 1) {
  n <- nrow(data_matrix)
  n_features <- ncol(data_matrix)
  X <- list()
  y <- numeric()
  
  for (i in 1:(n - window_size)) {
    X[[i]] <- data_matrix[i:(i + window_size - 1), ]
    y[i] <- data_matrix[i + window_size, target_col]
  }
  
  # Convert to 3D array: (samples, timesteps, features)
  X_array <- array(as.numeric(unlist(X)), 
                   dim = c(length(X), window_size, n_features))
  y_array <- as.numeric(y)
  
  storage.mode(X_array) <- "double"
  storage.mode(y_array) <- "double"
  
  list(X = X_array, y = y_array)
}

# Build multivariate models
build_multivariate_rnn <- function(window_size, n_features, units = 64) {
  model <- keras_model_sequential()
  model$add(keras$layers$Input(shape = as.integer(c(window_size, n_features))))
  model$add(keras$layers$SimpleRNN(units = as.integer(units)))
  model$add(keras$layers$Dropout(rate = 0.2))
  model$add(keras$layers$Dense(units = 1L))
  return(model)
}

build_multivariate_gru <- function(window_size, n_features, units = 64) {
  model <- keras_model_sequential()
  model$add(keras$layers$Input(shape = as.integer(c(window_size, n_features))))

  model$add(keras$layers$Dropout(rate = 0.2))
  model$add(keras$layers$Dense(units = 1L))
  return(model)
}

The Multivariate Paradox

The results reveal a counterintuitive finding: adding food price variables as inputs degraded deep learning performance while dramatically improving traditional models. SARIMAX achieved 0.063 RMSE with multivariate inputs—48% better than the best univariate model (LSTM at 0.121). Meanwhile, all deep learning architectures performed worse with additional variables: LSTM increased from 0.121 to 0.175, GRU from 0.164 to 0.219, and RNN from 0.133 to 0.316.

Why Deep Learning Failed in Multivariate Settings

Sample Size Constraints: With only 180 observations split across 6 features, deep learning models faced a curse of dimensionality. Each additional feature exponentially increased the parameter space that needed learning, but our training set (117 samples) remained fixed. The models likely overfit to spurious correlations in the training data rather than learning genuine relationships between food prices and CPI.

Traditional Models’ Structural Advantage: SARIMAX explicitly models linear relationships with regularization through statistical estimation, requiring fewer observations to fit reliably. Its superior performance (0.063 vs 0.152 for univariate ARIMA) confirms that food prices contain genuine predictive information—traditional methods just extract it more efficiently with limited data.

Model Selection for Real-World Deployment

For production CPI forecasting, deploy SARIMAX as the primary model. Its 48% error reduction over univariate methods justifies the added complexity, while its statistical framework provides interpretable coefficients showing how egg, beef, and milk prices influence overall inflation. Use univariate LSTM as a secondary monitor: if SARIMAX degrades relative to LSTM, it signals potential structural breaks in food-CPI relationships.

Practical Implications of Univariate-Only Analysis

Critical missed opportunity: Relying solely on univariate models would have yielded 0.121 RMSE (LSTM) while leaving the superior 0.063 SARIMAX undiscovered—effectively accepting 91% higher forecast errors. For policy applications where 0.06 RMSE translates to ±6% inflation forecast uncertainty, this difference is substantial. A central bank using univariate-only forecasts would make systematically worse interest rate decisions due to noisier inflation signals.

Strategic insight for forecasters: This comparison teaches that model sophistication must match problem structure, not just data volume. Multivariate economic data often contains explicit causal relationships (food prices → CPI) that traditional models capture parsimoniously, while deep learning requires massive datasets to discover these patterns empirically. When explaining to stakeholders: “More data helps, but only if your model can efficiently extract the signal—SARIMAX did this with 6 variables and 180 observations; deep learning would need 1000+ observations to match it.”

The lesson: Question whether your problem needs deep learning’s flexibility or benefits from traditional methods’ structural constraints. For CPI forecasting with established economic relationships, structure wins.
